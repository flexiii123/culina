<?php
/**
 * Datenbank-Konfiguration
 */

class Database {
    private $host = "localhost";
    private $db_name = "culina_db";
    private $username = "root";
    private $password = "";
    public $conn;

    public function getConnection() {
        $this->conn = null;

        try {
            $this->conn = new PDO(
                "mysql:host=" . $this->host . ";dbname=" . $this->db_name,
                $this->username,
                $this->password
            );
            $this->conn->exec("set names utf8");
            $this->conn->setAttribute(PDO::ATTR_ERRMODE, PDO::ERRMODE_EXCEPTION);
        } catch(PDOException $exception) {
            echo "Verbindungsfehler: " . $exception->getMessage();
        }

        return $this->conn;
    }
}

DATEI: config/schema.sql

-- Datenbank erstellen
CREATE DATABASE IF NOT EXISTS culina_db CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE culina_db;

-- Benutzer-Tabelle
CREATE TABLE IF NOT EXISTS users (
    id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    email VARCHAR(100) NOT NULL UNIQUE,
    password VARCHAR(255) NOT NULL,
    avatar VARCHAR(255) DEFAULT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    INDEX idx_email (email)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- Rezepte-Tabelle
CREATE TABLE IF NOT EXISTS recipes (
    id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT NOT NULL,
    title VARCHAR(200) NOT NULL,
    description TEXT NOT NULL,
    image VARCHAR(255) NOT NULL,
    cook_time VARCHAR(50) NOT NULL,
    difficulty ENUM('Einfach', 'Mittel', 'Schwer') NOT NULL,
    ingredients JSON NOT NULL,
    instructions JSON NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
    INDEX idx_user_id (user_id),
    INDEX idx_created_at (created_at)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- Bewertungen-Tabelle
CREATE TABLE IF NOT EXISTS ratings (
    id INT AUTO_INCREMENT PRIMARY KEY,
    recipe_id INT NOT NULL,
    user_id INT NOT NULL,
    rating INT NOT NULL CHECK (rating BETWEEN 1 AND 5),
    comment TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (recipe_id) REFERENCES recipes(id) ON DELETE CASCADE,
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
    UNIQUE KEY unique_user_recipe (recipe_id, user_id),
    INDEX idx_recipe_id (recipe_id),
    INDEX idx_user_id (user_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- Sessions-Tabelle (optional, für erweiterte Session-Verwaltung)
CREATE TABLE IF NOT EXISTS sessions (
    id VARCHAR(128) PRIMARY KEY,
    user_id INT NOT NULL,
    data TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    expires_at TIMESTAMP NOT NULL,
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
    INDEX idx_expires (expires_at)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

# Apache .htaccess Konfiguration für Culina Backend

# PHP Einstellungen
php_value upload_max_filesize 5M
php_value post_max_size 10M
php_value max_execution_time 60
php_value max_input_time 60

# Fehleranzeige (in Produktion ausschalten!)
php_flag display_errors Off
php_flag log_errors On

# CORS aktivieren
<IfModule mod_headers.c>
    Header set Access-Control-Allow-Origin "*"
    Header set Access-Control-Allow-Methods "POST, GET, PUT, DELETE, OPTIONS"
    Header set Access-Control-Allow-Headers "Content-Type, Authorization, X-Requested-With"
</IfModule>

# OPTIONS-Requests erlauben (für CORS-Preflight)
RewriteEngine On
RewriteCond %{REQUEST_METHOD} OPTIONS
RewriteRule ^(.*)$ $1 [R=200,L]

# Verzeichnis-Browsing deaktivieren
Options -Indexes

# Zugriff auf sensible Dateien verhindern
<FilesMatch "^\.">
    Order allow,deny
    Deny from all
</FilesMatch>

# .htaccess und andere Konfigurationsdateien schützen
<Files ~ "^(\.htaccess|\.htpasswd|config\.php|database\.php)$">
    Order allow,deny
    Deny from all
</Files>

# JSON-Dateien korrekt ausliefern
<IfModule mod_mime.c>
    AddType application/json .json
</IfModule>

# Gzip-Kompression aktivieren (optional)
<IfModule mod_deflate.c>
    AddOutputFilterByType DEFLATE application/json
    AddOutputFilterByType DEFLATE text/plain
    AddOutputFilterByType DEFLATE text/html
</IfModule>

==============================================
INSTALLATION
==============================================

1. Erstellen Sie folgende Ordnerstruktur:

culina-backend/
├── api/
│   ├── auth/
│   ├── recipes/
│   └── ratings/
├── config/
├── utils/
└── uploads/

2. Kopieren Sie die Dateien in die entsprechenden Ordner

3. MySQL-Datenbank einrichten:
   mysql -u root -p < config/schema.sql

4. Datenbank-Zugangsdaten in config/database.php anpassen

5. Server starten:
   php -S localhost:8000
